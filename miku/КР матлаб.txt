disp("Введте размерность матрицы: ")
n = input('')
A = randi([0,100],n)
B = [];
for i=1:n
	for j=1:n
		B(i,j)=sin(i)*sin(j);
	end
end

MA=1000;
for i=1:n
	for j=1:n
		if A(i,j)<MA
		    MA=A(i,j);
		end
	end
end

MB=1000;
for i=1:n
	for j=1:n
		if B(i,j)<MB
		    MB=B(i,j);
		end
	end
end

C=zeros(n);
for i=1:n
	for j=1:n
		for k=1:n
		    C(i,k)=A(i,:)*B(:,k);
		end
	end
end

for i=1:n
	for j=1:n
		C(i,j)=C(i,j)*MA*MB;
	end
end

function ZeroPlot2(f)
	x=[-5,5];
	fplot(f,x,'b-', 'LineWidth', 1);
	grid on;
	xlabel('x');
	ylabel('f(x)');
	x_initial = [-5, -1, 1, 5]; hold on;
	roots = [];
	
	for i = 1:length(x_initial)
	    root = fzero(f, x_initial(i));
	    if root >= -5 && root <= 5
	        roots = [roots, root];
	    end
	end
	roots = unique(roots)
	for i = 1:length(roots)
	    plot(roots(i), 0, 'ro', 'MarkerSize', 8, 'MarkerFaceColor', 'r');
	end
end

function ZeroPlot3(f)
	x1_range = -5:0.1:5;
	x2_range = -5:0.1:5;
	[X1, X2] = meshgrid(x1_range, x2_range);
	Z = zeros(size(X1));
	for i = 1:numel(X1)
	    Z(i) = f([X1(i), X2(i)]);
	end
	figure();
	surf(X1, X2, Z);
	xlabel('x1');
	ylabel('x2');
	zlabel('f(x1, x2)'); hold on;
	[xy_min, f_min] = fminsearch(f, [0, 0])
	plot3(xy_min(1), xy_min(2), f_min, 'ro');
end


f1 = @(x) 5*cos(x);
ZeroPlot2(f1)

f2 = @(x) abs(x(1)-1) + abs(x(2)-5);
